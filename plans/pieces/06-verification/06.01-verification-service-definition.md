# 06.01 — Verification Service Definition and Domain Contracts

**Set:** 06-verification · **Queue:** 1 · **Depends on:** none
**Status:** todo
**Harness primitive:** Service Definition · **Package:** `@deepseek-ai/dsh-verification`

## Summary

Defines the abstract `VerificationStore` service class extending `@deepseek-ai/cordis.Service` and merges `Context.verification` and typed verification events into Cordis. Establishes branded types (`VerificationEventId`) and core domain interfaces (`VerificationEvent`, `VerificationState`, `WorkspaceVerificationStatus`, `PinnedSuiteManifest`, `RedGateFailureClassification`, `RedGateVerificationRequest`, `RedGateVerificationResult`, `GreenGateVerificationRequest`, `GreenGateVerificationResult`). Governs append-only attributable evidence recording, gate evaluation contracts, and workspace freshness across the host and worker planes.

## Behaviour

- **Given** the Cordis root container **When** `@deepseek-ai/dsh-verification` is loaded **Then** `ctx.verification` is recognized on `Context` as an instance of `VerificationStore`.
- **Given** an arbitrary positive integer **When** `asVerificationEventId` is invoked **Then** it validates and returns a branded `VerificationEventId`.
- **Given** a non-integer or negative number **When** `asVerificationEventId` is invoked **Then** it throws a descriptive `TypeError`.
- **Given** an active `VerificationStore` **When** verification events or gate executions complete **Then** typed events (`verification/event-recorded`, `verification/red-gate`, `verification/green-gate`, `verification/state-dirtied`) dispatch on the Cordis event bus.
- **Given** a card request for gate evaluation **When** `evaluateRedGate` or `evaluateGreenGate` is called **Then** signatures enforce immutable requests and structured result payloads.

## Harness fit

This is a `Service Definition` primitive owned by `@deepseek-ai/dsh-verification`. It declares the abstract `VerificationStore` interface and merges `interface Context { verification: VerificationStore }` into Cordis (`packages/core/tools/src/index.ts:44-47`). Isolating domain contracts from concrete storage and execution engines allows downstream modules (`@deepseek-ai/dsh-verification-stop`, `@deepseek-ai/dsh-guard-test-pinning`, `@deepseek-ai/dsh-axiom-verifier`) to bind to pure interfaces without coupling to storage engines or subprocess runners.

## Contracts

```typescript
export type VerificationEventId = number & { readonly __brand: unique symbol }
export type VerificationKind = 'test' | 'typecheck' | 'lint' | 'build' | 'format' | 'red_gate' | 'green_gate'
export type VerificationScope = 'workspace' | 'package' | 'file' | 'ad_hoc'
export type VerificationStatus = 'passed' | 'failed' | 'error'

export interface VerificationEvent {
  readonly id: VerificationEventId
  readonly createdAt: string
  readonly sessionId: string
  readonly cwd: string
  readonly root: string
  readonly command: string
  readonly canonicalCommand: string
  readonly kind: VerificationKind
  readonly scope: VerificationScope
  readonly status: VerificationStatus
  readonly exitCode: number
  readonly outputSummary: string
  readonly metadata?: Readonly<Record<string, unknown>> | undefined
}

export interface VerificationState {
  readonly sessionId: string
  readonly root: string
  readonly lastEventId?: VerificationEventId | undefined
  readonly lastEditAt?: string | undefined
  readonly changedPaths: readonly string[]
}

export interface WorkspaceVerificationStatus {
  readonly status: 'passed' | 'failed' | 'stale' | 'unverified'
  readonly latestEvent?: VerificationEvent | undefined
  readonly unverifiedChanges: readonly string[]
}

export interface PinnedSuiteManifest {
  readonly cardId: string
  readonly baseCommit: string
  readonly pinnedAt: string
  readonly aggregateHash: string
  readonly files: Readonly<Record<string, string>>
  readonly testRunnerConfigs: Readonly<Record<string, string>>
}

export type RedGateFailureClassification =
  | 'EXPECTED_ASSERTION_FAILURE'
  | 'EXPECTED_MISSING_SYMBOL'
  | 'EXPECTED_NOT_IMPLEMENTED'
  | 'SETUP_SYNTAX_ERROR'
  | 'SETUP_IMPORT_ERROR'
  | 'SETUP_HARNESS_ERROR'
  | 'SETUP_ZERO_TESTS'
  | 'UNEXPECTED_PASS'

export interface RedGateVerificationRequest {
  readonly cardId: string
  readonly worktreeRoot: string
  readonly baseCommit: string
  readonly testCommand: string
  readonly testGlobs: readonly string[]
  readonly configGlobs?: readonly string[] | undefined
  readonly signal?: AbortSignal | undefined
}

export interface RedGateVerificationResult {
  readonly passed: boolean
  readonly classification: RedGateFailureClassification
  readonly exitCode: number
  readonly outputSummary: string
  readonly manifest?: PinnedSuiteManifest | undefined
  readonly eventId?: VerificationEventId | undefined
  readonly failureReason?: string | undefined
}

export interface GreenGateVerificationRequest {
  readonly cardId: string
  readonly worktreeRoot: string
  readonly testCommand: string
  readonly pinnedManifest: PinnedSuiteManifest
  readonly signal?: AbortSignal | undefined
}

export interface GreenGateVerificationResult {
  readonly passed: boolean
  readonly exitCode: number
  readonly hashMatch: boolean
  readonly outputSummary: string
  readonly eventId?: VerificationEventId | undefined
  readonly antiCheatViolations: readonly string[]
}

export abstract class VerificationStore extends Service {
  constructor(ctx: Context) { super(ctx, 'verification') }
  abstract recordEvent(event: Omit<VerificationEvent, 'id' | 'createdAt'>): Promise<VerificationEvent>
  abstract getStatus(sessionId: string, cwd: string): Promise<WorkspaceVerificationStatus>
  abstract markWorkspaceEdited(sessionId: string, root: string, paths: readonly string[]): Promise<void>
  abstract isExitStatusAttributable(command: string): boolean
  abstract evaluateRedGate(request: RedGateVerificationRequest): Promise<RedGateVerificationResult>
  abstract evaluateGreenGate(request: GreenGateVerificationRequest): Promise<GreenGateVerificationResult>
  abstract getPinnedManifest(cardId: string): Promise<PinnedSuiteManifest | undefined>
}

declare module '@deepseek-ai/cordis' {
  interface Context { verification: VerificationStore }
  interface Events {
    'verification/event-recorded'(event: VerificationEvent): void
    'verification/red-gate'(result: RedGateVerificationResult): void
    'verification/green-gate'(result: GreenGateVerificationResult): void
    'verification/state-dirtied'(state: VerificationState): void
  }
}
```
